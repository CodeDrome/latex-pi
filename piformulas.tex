\section{Francois Viete}

\[
\frac{2}{\pi} = 
\frac{\sqrt{2}}{2} \cdot 
\frac{\sqrt{2+\sqrt{2}}}{2} \cdot
\frac{\sqrt{2+\sqrt{2+\sqrt{2}}}}{2} 
\cdots
\]

\[
\pi = (1 \div \frac{2}{\pi}) \times 2
\]

\[
\pi = \left( 1 \div \frac{2}{\pi} \right) \times 2
\]
